19/// General text style
20#[derive(Debug, Clone)]
21pub struct TextVisuals {
22    /// Font used
23    pub font_id: FontId,
24
25    /// Font color
26    pub color: Color32,
27}
28
29impl TextVisuals {
30    /// Text in `color`, using the given font.
31    ///
32    /// `style.override_font_id` wins over `font`, if it is set.
33    pub fn new(style: &Style, font: impl Into<FontSelection>, color: Color32) -> Self {
34        Self {
35            color,
36            font_id: style
37                .override_font_id
38                .clone()
39                .unwrap_or_else(|| font.into().resolve(style)),
40        }
41    }
42
43    /// The text of a widget, colored by the [`WidgetVisuals`] of its current state.
44    pub fn from_widget_visuals(
45        style: &Style,
46        font: impl Into<FontSelection>,
47        widget_visuals: &WidgetVisuals,
48    ) -> Self {
49        Self::new(style, font, widget_visuals.text_color())
50    }
51}

53/// Visual and layout style shared by widgets built from a [`WidgetAtom`].
54#[derive(Debug, Clone)]
55pub struct AtomLayoutStyle {
56    /// Alignment of the atoms within the allocated rectangle.
57    ///
58    /// `None` uses the alignment of the surrounding [`crate::Ui`].
59    pub align2: Option<Align2>,
60
61    /// Minimum size of the atom layout.
62    pub min_size: Vec2,
63
64    /// Space between adjacent atoms.
65    pub gap: f32,
66
67    /// Frame around the atoms.
68    pub frame: Frame,
69
70    /// Fallback visuals for text atoms.
71    pub text_style: TextVisuals,
72
73    /// Fallback tint for images whose tint is [`Color32::WHITE`] (untinted).
74    pub image_tint: Color32,
75}
76
77impl Default for AtomLayoutStyle {
78    fn default() -> Self {
79        Self {
80            align2: None,
81            min_size: Vec2::ZERO,
82            gap: 0.0,
83            frame: Frame::default(),
84            text_style: TextVisuals {
85                font_id: FontId::default(),
86                color: Color32::WHITE,
87            },
88            image_tint: Color32::WHITE,
89        }
90    }
91}
92
93impl AtomLayoutStyle {
94    /// Apply this style to a [`WidgetAtom`].
95    ///
96    /// A per-widget [`WidgetAtom::gap`] wins over [`Self::gap`], so widgets like
97    /// [`crate::DragValue`] can pack their atoms tighter than the theme does.
98    pub fn apply(self, mut layout: WidgetAtom<'_>) -> WidgetAtom<'_> {
99        let Self {
100            align2,
101            min_size,
102            gap,
103            frame,
104            text_style,
105            image_tint,
106        } = self;
107        layout.map_images(|image| {
108            let current_tint = image.image_options().tint;
109            // Multiply the tints so they are combined
110            image.tint(current_tint * image_tint)
111        });
112
113        let layout = layout
114            .min_size(min_size)
115            .fallback_gap(gap)
116            .frame(frame)
117            .fallback_font(text_style.font_id)
118            .fallback_text_color(text_style.color);
119
120        if let Some(align2) = align2 {
121            layout.align2(align2)
122        } else {
123            layout
124        }
125    }
126}

189/// The different state of a widget can be
190#[derive(Default, Clone, Copy, Debug, PartialEq, Eq, Hash)]
191pub enum WidgetState {
192    Noninteractive,
193    #[default]
194    Inactive,
195    Hovered,
196    Active,
197}
198
199impl Widgets {
200    /// The widget visuals according to the state
201    pub fn state(&self, state: WidgetState) -> &WidgetVisuals {
202        match state {
203            WidgetState::Noninteractive => &self.noninteractive,
204            WidgetState::Inactive => &self.inactive,
205            WidgetState::Hovered => &self.hovered,
206            WidgetState::Active => &self.active,
207        }
208    }
209}
210
211impl Response {
212    pub fn widget_state(&self) -> WidgetState {
213        if !self.sense.interactive() {
214            WidgetState::Noninteractive
215        } else if self.is_pointer_button_down_on() || self.has_focus() || self.clicked() {
216            WidgetState::Active
217        } else if self.hovered() || self.highlighted() {
218            WidgetState::Hovered
219        } else {
220            WidgetState::Inactive
221        }
222    }
223}
