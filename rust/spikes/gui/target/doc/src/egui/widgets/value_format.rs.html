1//! How a number is written, read back, and applied while being edited.
2//!
3//! Used by [`crate::DragValue`], [`crate::Slider`], and [`crate::RangeSlider`].
4
5use alloc::sync::Arc;
6use core::ops::RangeInclusive;
7
8use crate::{Atoms, IntoAtoms, MINUS_CHAR_STR, emath};
9
10/// Formats a value for display, given the range of decimals it may show.
11pub type NumFormatter<'a> = Arc<dyn 'a + Fn(f64, RangeInclusive<usize>) -> String>;
12
13/// Parses what the user typed, returning `None` if it is not a number.
14pub type NumParser<'a> = Arc<dyn 'a + Fn(&str) -> Option<f64>>;

18/// How a number is written, read back, and applied while being edited.
19///
20/// Owned by [`crate::DragValue`], and by [`crate::Slider`] and [`crate::RangeSlider`]
21/// for the numbers they show beside the rail.
22#[derive(Clone)]
23pub struct ValueFormat<'a> {
24    /// Shown before the number, e.g. `"x: "`.
25    pub prefix: Atoms<'a>,
26
27    /// Shown after the number, e.g. a unit.
28    pub suffix: Atoms<'a>,
29
30    /// Show at least this many decimals.
31    pub min_decimals: usize,
32
33    /// Show at most this many decimals, and round values to it.
34    pub max_decimals: Option<usize>,
35
36    /// Turns a value into text. Default: [`crate::style::NumberFormatter`].
37    pub custom_formatter: Option<NumFormatter<'a>>,
38
39    /// Turns text back into a value. Default: accepts plain decimal numbers.
40    pub custom_parser: Option<NumParser<'a>>,
41
42    /// Apply each keystroke while typing, rather than on enter.
43    pub update_while_editing: bool,
44}
45
46impl Default for ValueFormat<'_> {
47    fn default() -> Self {
48        Self {
49            prefix: Atoms::default(),
50            suffix: Atoms::default(),
51            min_decimals: 0,
52            max_decimals: None,
53            custom_formatter: None,
54            custom_parser: None,
55            update_while_editing: true,
56        }
57    }
58}
59
60impl<'a> ValueFormat<'a> {
61    /// Show a prefix before the number, e.g. "x: ".
62    ///
63    /// Goes in front of any prefix already set, so `.prefix("b").prefix("a")` shows `ab`.
64    #[inline]
65    pub fn prefix(mut self, prefix: impl IntoAtoms<'a>) -> Self {
66        self.prefix.extend_left(prefix.into_atoms());
67        self
68    }
69
70    /// Add a suffix to the number, this can be e.g. a unit ("°" or " m").
71    ///
72    /// Goes after any suffix already set, so `.suffix("a").suffix("b")` shows `ab`.
73    #[inline]
74    pub fn suffix(mut self, suffix: impl IntoAtoms<'a>) -> Self {
75        self.suffix.extend_right(suffix.into_atoms());
76        self
77    }
78
79    /// Set a minimum number of decimals to display.
80    #[inline]
81    pub fn min_decimals(mut self, min_decimals: usize) -> Self {
82        self.min_decimals = min_decimals;
83        self
84    }
85
86    /// Set a maximum number of decimals to display.
87    /// Values will also be rounded to this number of decimals.
88    #[inline]
89    pub fn max_decimals(mut self, max_decimals: usize) -> Self {
90        self.max_decimals = Some(max_decimals);
91        self
92    }
93
94    /// Set the maximum number of decimals to display, or `None` to let the widget pick.
95    ///
96    /// Values will also be rounded to this number of decimals when it is set.
97    #[inline]
98    pub fn max_decimals_opt(mut self, max_decimals: Option<usize>) -> Self {
99        self.max_decimals = max_decimals;
100        self
101    }
102
103    /// Set an exact number of decimals to display.
104    /// Values will also be rounded to this number of decimals.
105    #[inline]
106    pub fn fixed_decimals(mut self, num_decimals: usize) -> Self {
107        self.min_decimals = num_decimals;
108        self.max_decimals = Some(num_decimals);
109        self
110    }
111
112    /// Set custom formatter defining how numbers are converted into text.
113    ///
114    /// A custom formatter takes a `f64` for the numeric value and a `RangeInclusive<usize>` representing
115    /// the decimal range i.e. minimum and maximum number of decimal places shown.
116    ///
117    /// See also: [`Self::custom_parser`]
118    pub fn custom_formatter(
119        mut self,
120        formatter: impl 'a + Fn(f64, RangeInclusive<usize>) -> String,
121    ) -> Self {
122        self.custom_formatter = Some(Arc::new(formatter));
123        self
124    }
125
126    /// Set custom parser defining how the text input is parsed into a number.
127    ///
128    /// A custom parser takes an `&str` to parse into a number and returns a `f64` if it was successfully parsed
129    /// or `None` otherwise.
130    ///
131    /// See also: [`Self::custom_formatter`]
132    #[inline]
133    pub fn custom_parser(mut self, parser: impl 'a + Fn(&str) -> Option<f64>) -> Self {
134        self.custom_parser = Some(Arc::new(parser));
135        self
136    }
137
138    /// Set `custom_formatter` and `custom_parser` to display and parse numbers as binary integers. Floating point
139    /// numbers are *not* supported.
140    ///
141    /// `min_width` specifies the minimum number of displayed digits; if the number is shorter than this, it will be
142    /// prefixed with additional 0s to match `min_width`.
143    ///
144    /// If `twos_complement` is true, negative values will be displayed as the 2's complement representation. Otherwise
145    /// they will be prefixed with a '-' sign.
146    ///
147    /// # Panics
148    ///
149    /// Panics if `min_width` is 0.
150    pub fn binary(self, min_width: usize, twos_complement: bool) -> Self {
151        assert!(
152            0 < min_width,
153            "ValueFormat::binary: `min_width` must be greater than 0"
154        );
155        if twos_complement {
156            self.custom_formatter(move |n, _| format!("{:0>min_width$b}", n as i64))
157        } else {
158            self.custom_formatter(move |n, _| {
159                let sign = if n < 0.0 { MINUS_CHAR_STR } else { "" };
160                format!("{sign}{:0>min_width$b}", n.abs() as i64)
161            })
162        }
163        .custom_parser(|s| i64::from_str_radix(s, 2).map(|n| n as f64).ok())
164    }
165
166    /// Set `custom_formatter` and `custom_parser` to display and parse numbers as octal integers. Floating point
167    /// numbers are *not* supported.
168    ///
169    /// `min_width` specifies the minimum number of displayed digits; if the number is shorter than this, it will be
170    /// prefixed with additional 0s to match `min_width`.
171    ///
172    /// If `twos_complement` is true, negative values will be displayed as the 2's complement representation. Otherwise
173    /// they will be prefixed with a '-' sign.
174    ///
175    /// # Panics
176    ///
177    /// Panics if `min_width` is 0.
178    pub fn octal(self, min_width: usize, twos_complement: bool) -> Self {
179        assert!(
180            0 < min_width,
181            "ValueFormat::octal: `min_width` must be greater than 0"
182        );
183        if twos_complement {
184            self.custom_formatter(move |n, _| format!("{:0>min_width$o}", n as i64))
185        } else {
186            self.custom_formatter(move |n, _| {
187                let sign = if n < 0.0 { MINUS_CHAR_STR } else { "" };
188                format!("{sign}{:0>min_width$o}", n.abs() as i64)
189            })
190        }
191        .custom_parser(|s| i64::from_str_radix(s, 8).map(|n| n as f64).ok())
192    }
193
194    /// Set `custom_formatter` and `custom_parser` to display and parse numbers as hexadecimal integers. Floating point
195    /// numbers are *not* supported.
196    ///
197    /// `min_width` specifies the minimum number of displayed digits; if the number is shorter than this, it will be
198    /// prefixed with additional 0s to match `min_width`.
199    ///
200    /// If `twos_complement` is true, negative values will be displayed as the 2's complement representation. Otherwise
201    /// they will be prefixed with a '-' sign.
202    ///
203    /// # Panics
204    ///
205    /// Panics if `min_width` is 0.
206    pub fn hexadecimal(self, min_width: usize, twos_complement: bool, upper: bool) -> Self {
207        assert!(
208            0 < min_width,
209            "ValueFormat::hexadecimal: `min_width` must be greater than 0"
210        );
211        match (twos_complement, upper) {
212            (true, true) => {
213                self.custom_formatter(move |n, _| format!("{:0>min_width$X}", n as i64))
214            }
215            (true, false) => {
216                self.custom_formatter(move |n, _| format!("{:0>min_width$x}", n as i64))
217            }
218            (false, true) => self.custom_formatter(move |n, _| {
219                let sign = if n < 0.0 { MINUS_CHAR_STR } else { "" };
220                format!("{sign}{:0>min_width$X}", n.abs() as i64)
221            }),
222            (false, false) => self.custom_formatter(move |n, _| {
223                let sign = if n < 0.0 { MINUS_CHAR_STR } else { "" };
224                format!("{sign}{:0>min_width$x}", n.abs() as i64)
225            }),
226        }
227        .custom_parser(|s| i64::from_str_radix(s, 16).map(|n| n as f64).ok())
228    }
229
230    /// Update the value on each key press when text-editing the value.
231    ///
232    /// Default: `true`.
233    /// If `false`, the value will only be updated when user presses enter or deselects the value.
234    #[inline]
235    pub fn update_while_editing(mut self, update: bool) -> Self {
236        self.update_while_editing = update;
237        self
238    }
239
240    /// Rounds `value` to `max_decimals`, if set.
241    pub fn round(&self, value: f64) -> f64 {
242        match self.max_decimals {
243            Some(max_decimals) => emath::round_to_decimals(value, max_decimals),
244            None => value,
245        }
246    }
247}
