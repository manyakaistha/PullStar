22/// Apply a menu style to the [`Style`].
23///
24/// Mainly removes the background stroke and the inactive background fill.
25pub fn menu_style(style: &mut Style) {
26    style.spacing.button_padding = vec2(2.0, 0.0);
27    style.visuals.widgets.active.bg_stroke = Stroke::NONE;
28    style.visuals.widgets.open.bg_stroke = Stroke::NONE;
29    style.visuals.widgets.hovered.bg_stroke = Stroke::NONE;
30    style.visuals.widgets.inactive.weak_bg_fill = Color32::TRANSPARENT;
31    style.visuals.widgets.inactive.bg_stroke = Stroke::NONE;
32}

34/// Find the root [`UiStack`] of the menu.
35pub fn find_menu_root(ui: &Ui) -> &UiStack {
36    ui.stack()
37        .iter()
38        .find(|stack| {
39            stack.is_root_ui()
40                || [Some(UiKind::Popup), Some(UiKind::Menu)].contains(&stack.kind())
41                || stack.info.tags.contains(MenuConfig::MENU_CONFIG_TAG)
42        })
43        .expect("We should always find the root")
44}
45
46/// Find the [`UiStack`] of the [`MenuBar`] this [`Ui`] is directly part of, if any.
47///
48/// Returns `None` if the closest menu root is not a menu bar (e.g. if we are in a popup menu).
49fn find_menu_bar_root(ui: &Ui) -> Option<&UiStack> {
50    let root = find_menu_root(ui);
51    root.info
52        .tags
53        .get_downcast::<MenuConfig>(MenuConfig::MENU_CONFIG_TAG)
54        .is_some_and(|config| config.bar)
55        .then_some(root)
56}

58/// Is this Ui part of a menu?
59///
60/// Returns `false` if this is a menu bar.
61/// Should be used to determine if we should show a menu button or submenu button.
62pub fn is_in_menu(ui: &Ui) -> bool {
63    for stack in ui.stack().iter() {
64        if let Some(config) = stack
65            .info
66            .tags
67            .get_downcast::<MenuConfig>(MenuConfig::MENU_CONFIG_TAG)
68        {
69            return !config.bar;
70        }
71        if [Some(UiKind::Popup), Some(UiKind::Menu)].contains(&stack.kind()) {
72            return true;
73        }
74    }
75    false
76}

78/// Configuration and style for menus.
79#[derive(Clone, Debug)]
80pub struct MenuConfig {
81    /// Is this a menu bar?
82    bar: bool,
83
84    /// If the user clicks, should we close the menu?
85    pub close_behavior: PopupCloseBehavior,
86
87    /// Override the menu style.
88    ///
89    /// Default is [`menu_style`].
90    pub style: StyleModifier,
91
92    /// The visual gap between the frame of a menu and the frames of the submenus it opens.
93    ///
94    /// `0.0` places submenus flush against their parent menu.
95    ///
96    /// Default is `2.0`.
97    pub submenu_gap: f32,
98}

100impl Default for MenuConfig {
101    fn default() -> Self {
102        Self {
103            close_behavior: PopupCloseBehavior::default(),
104            bar: false,
105            style: menu_style.into(),
106            submenu_gap: 2.0,

108    }
109}

168/// Holds the state of the menu.
169#[derive(Clone)]
170pub struct MenuState {
171    /// The currently open sub menu in this menu.
172    pub open_item: Option<Id>,
173    last_visible_pass: u64,
174    focus_first: bool,
175
176    /// A stationary pointer should not close a submenu opened with the keyboard.
177    ignore_hover: bool,
178
179    /// The widget that arrow keys moved focus to in the last pass.
180    arrow_focus: Option<Id>,
181}
182
183impl MenuState {
184    pub const ID: &'static str = "menu_state";
185
186    /// Find the root of the menu and get the state
187    pub fn from_ui<R>(ui: &Ui, f: impl FnOnce(&mut Self, &UiStack) -> R) -> R {
188        let stack = find_menu_root(ui);
189        Self::from_id(ui.ctx(), stack.unique_id, |state| f(state, stack))
190    }
191
192    /// Get the state via the menus root [`Ui`] id
193    pub fn from_id<R>(ctx: &Context, id: Id, f: impl FnOnce(&mut Self) -> R) -> R {
194        let pass_nr = ctx.cumulative_pass_nr();
195        ctx.data_mut(|data| {
196            let state_id = id.with(Self::ID);
197            let mut state = data.get_temp(state_id).unwrap_or(Self {
198                open_item: None,
199                last_visible_pass: pass_nr,
200                focus_first: false,
201                ignore_hover: false,
202                arrow_focus: None,
203            });
204            // If the menu was closed for at least a frame, reset the open item
205            if state.last_visible_pass + 1 < pass_nr {
206                state.open_item = None;
207                state.focus_first = false;
208                state.ignore_hover = false;
209                state.arrow_focus = None;
210            }
211            if let Some(item) = state.open_item
212                && data
213                    .get_temp(item.with(Self::ID))
214                    .is_none_or(|item: Self| item.last_visible_pass + 1 < pass_nr)
215            {
216                // If the open item wasn't shown for at least a frame, reset the open item
217                state.open_item = None;
218            }
219            let r = f(&mut state);
220            data.insert_temp(state_id, state);
221            r
222        })
223    }
224
225    pub fn mark_shown(ctx: &Context, id: Id) {
226        let pass_nr = ctx.cumulative_pass_nr();
227        Self::from_id(ctx, id, |state| {
228            state.last_visible_pass = pass_nr;
229        });
230    }
231
232    /// Navigate within this menu after all of its entries have registered focus interest.
233    pub(crate) fn handle_keyboard(ui: &Ui, anchor_widget: Option<Id>) {
234        let menu_id = ui.layer_id().id;
235        let (allows_interaction, direction) =
236            ui.memory(|mem| (mem.allows_interaction(ui.layer_id()), mem.focus_direction()));
237        let active = !ui.is_sizing_pass() && allows_interaction;
238        let focus_first = Self::from_id(ui.ctx(), menu_id, |state| {
239            state.arrow_focus = None;
240            active && core::mem::take(&mut state.focus_first)
241        });
242        if !active || (!focus_first && !direction.is_cardinal()) {
243            return;
244        }
245        let is_submenu = anchor_widget.is_some_and(|id| SubMenu::id_from_widget_id(id) == menu_id);
246        let focused = ui.memory(|mem| mem.focused());
247        let entries: Vec<Id> = ui.ctx().viewport(|viewport| {
248            viewport
249                .this_pass
250                .widgets
251                .focusable_widgets_in_layer(ui.layer_id())
252                .collect()
253        });
254        let focus_first = focus_first && focused == anchor_widget;
255        let index = entries.iter().position(|id| Some(*id) == focused);
256        let entering = focused == anchor_widget && !is_submenu;
257        let from_empty = focused.is_none() && Self::is_deepest_open_sub_menu(ui.ctx(), menu_id);
258        if index.is_none() && !entering && !from_empty && !focus_first {
259            return;
260        }
261        let next = if focus_first {
262            entries.first().copied()
263        } else {
264            match direction {
265                FocusDirection::Down => index
266                    .and_then(|index| entries.get(index + 1))
267                    .or_else(|| entries.first())
268                    .copied(),
269                FocusDirection::Up => index
270                    .and_then(|index| index.checked_sub(1))
271                    .and_then(|index| entries.get(index))
272                    .or_else(|| entries.last())
273                    .copied(),
274                FocusDirection::Left => anchor_widget,
275                // SubMenu handles Right on its own button before we get here.
276                _ => None,
277            }
278        };
279        Self::from_id(ui.ctx(), menu_id, |state| state.arrow_focus = next);
280        ui.memory_mut(|mem| {
281            mem.move_focus(FocusDirection::None);
282            if let Some(next) = next {
283                mem.request_focus_next_frame(next);
284            }
285        });
286        ui.request_repaint();
287    }
288
289    /// Is the menu with this id the deepest sub menu? (-> no child sub menu is open)
290    ///
291    /// Note: This only returns correct results if called after the menu contents were shown.
292    pub fn is_deepest_open_sub_menu(ctx: &Context, id: Id) -> bool {
293        let pass_nr = ctx.cumulative_pass_nr();
294        let open_item = Self::from_id(ctx, id, |state| state.open_item);
295        // If we have some open item, check if that was actually shown this frame
296        open_item.is_none_or(|submenu_id| {
297            Self::from_id(ctx, submenu_id, |state| state.last_visible_pass != pass_nr)
298        })
299    }
300}

302/// Horizontal menu bar where you can add [`MenuButton`]s.
303///
304/// The menu bar goes well in a [`crate::Panel::top`],
305/// but can also be placed in a [`crate::Window`].
306/// In the latter case you may want to wrap it in [`Frame`].
307///
308/// ### Example:
309/// ```
310/// # egui::__run_test_ui(|ui| {
311/// egui::MenuBar::new().ui(ui, |ui| {
312///     ui.menu_button("File", |ui| {
313///         if ui.button("Quit").clicked() {
314///             ui.send_viewport_cmd(egui::ViewportCommand::Close);
315///         }
316///     });
317/// });
318/// # });
319/// ```
320#[derive(Clone, Debug)]
321pub struct MenuBar {
322    config: MenuConfig,
323    style: StyleModifier,
324}
325
326impl Default for MenuBar {
327    fn default() -> Self {
328        Self {
329            config: MenuConfig::default(),
330            style: menu_style.into(),
331        }
332    }
333}
334
335impl MenuBar {
336    pub fn new() -> Self {
337        Self::default()
338    }
339
340    /// Set the style for buttons in the menu bar.
341    ///
342    /// Doesn't affect the style of submenus, use [`MenuConfig::style`] for that.
343    /// Default is [`menu_style`].
344    #[inline]
345    pub fn style(mut self, style: impl Into<StyleModifier>) -> Self {
346        self.style = style.into();
347        self
348    }
349
350    /// Set the config for submenus.
351    ///
352    /// Note: The config will only be passed when using [`MenuButton`], not via [`Popup::menu`].
353    #[inline]
354    pub fn config(mut self, config: MenuConfig) -> Self {
355        self.config = config;
356        self
357    }
358
359    /// Show the menu bar.
360    #[inline]
361    pub fn ui<R>(self, ui: &mut Ui, content: impl FnOnce(&mut Ui) -> R) -> InnerResponse<R> {
362        let Self { mut config, style } = self;
363        config.bar = true;
364        // TODO(lucasmerlin): It'd be nice if we had a ui.horizontal_builder or something
365        // So we don't need the nested scope here
366        ui.horizontal(|ui| {
367            ui.scope_builder(
368                UiBuilder::new()
369                    .layout(Layout::left_to_right(Align::Center))
370                    .ui_stack_info(
371                        UiStackInfo::new(UiKind::Menu)
372                            .with_tag_value(MenuConfig::MENU_CONFIG_TAG, config),
373                    ),
374                |ui| {
375                    style.apply(ui.style_mut());
376
377                    // Take full width and fixed height:
378                    let height = ui.spacing().interact_size.y;
379                    ui.set_min_size(vec2(ui.available_width(), height));
380
381                    content(ui)
382                },
383            )
384            .inner
385        })
386    }
387}

389/// A thin wrapper around a [`Button`] that shows a [`Popup::menu`] when clicked.
390///
391/// The only thing this does is search for the current menu config (if set via [`MenuBar`]).
392/// If your menu button is not in a [`MenuBar`] it's fine to use [`Ui::button`] and [`Popup::menu`]
393/// directly.
394pub struct MenuButton<'a> {
395    pub button: Button<'a>,
396    pub config: Option<MenuConfig>,
397}
398
399impl<'a> MenuButton<'a> {
400    /// Salt for the id (relative to the [`MenuBar`]) where we store the popup id of
401    /// the currently open menu in that bar.
402    const OPEN_MENU_ID_SALT: &'static str = "egui_menu_bar_open_menu";
403
404    pub fn new(atoms: impl IntoAtoms<'a>) -> Self {
405        Self::from_button(Button::new(atoms.into_atoms()))
406    }
407
408    /// Set the config for the menu.
409    #[inline]
410    pub fn config(mut self, config: MenuConfig) -> Self {
411        self.config = Some(config);
412        self
413    }
414
415    /// Create a new menu button from a [`Button`].
416    #[inline]
417    pub fn from_button(button: Button<'a>) -> Self {
418        Self {
419            button,
420            config: None,
421        }
422    }
423
424    /// Show the menu button.
425    pub fn ui<R>(
426        self,
427        ui: &mut Ui,
428        content: impl FnOnce(&mut Ui) -> R,
429    ) -> (Response, Option<InnerResponse<R>>) {
430        let response = self.button.ui(ui);
431        let mut config = self.config.unwrap_or_else(|| MenuConfig::find(ui));
432        config.bar = false;
433
434        let mut menu = Popup::menu(&response);
435        let popup_id = menu.get_id();
436
437        // If we are part of a menu bar, we remember which of its menus is open, so that
438        // hovering another menu button in the same bar can switch to that menu without
439        // requiring a click. This is the typical behavior of menu bars at the top of a window.
440        let bar_open_menu_id =
441            find_menu_bar_root(ui).map(|bar| bar.unique_id.with(Self::OPEN_MENU_ID_SALT));
442        if let Some(bar_open_menu_id) = bar_open_menu_id
443            && response.hovered()
444            && !response.clicked()
445        {
446            let open_menu_in_bar = ui.data(|d| d.get_temp::<Id>(bar_open_menu_id));
447            if let Some(open_menu_in_bar) = open_menu_in_bar
448                && open_menu_in_bar != popup_id
449                && Popup::is_id_open(ui.ctx(), open_menu_in_bar)
450            {
451                menu = menu.open_memory(Some(SetOpenCommand::Bool(true)));
452            }
453        }
454
455        let inner = menu
456            .close_behavior(config.close_behavior)
457            .style(config.style.clone())
458            .info(
459                UiStackInfo::new(UiKind::Menu).with_tag_value(MenuConfig::MENU_CONFIG_TAG, config),
460            )
461            .show(content);
462
463        if let Some(bar_open_menu_id) = bar_open_menu_id
464            && inner.is_some()
465        {
466            ui.data_mut(|d| d.insert_temp(bar_open_menu_id, popup_id));
467        }
468
469        (response, inner)
470    }
471}

473/// A submenu button that shows a [`SubMenu`] if a [`Button`] is hovered.
474pub struct SubMenuButton<'a> {
475    pub button: Button<'a>,
476    pub sub_menu: SubMenu,
477}
478
479impl<'a> SubMenuButton<'a> {
480    /// The submenu arrow triangle shape.
481    pub fn arrow_shape(rect: Rect, color: impl Into<Color32>) -> Shape {
482        let rect = Rect::from_center_size(
483            rect.center(),
484            vec2(rect.width() * 0.55, rect.height() * 0.35),
485        );
486        Shape::rotated_triangle(rect, -core::f32::consts::TAU / 4.0, color)
487    }
488
489    /// An [`Atom`] painting the [`Self::arrow_shape`].
490    ///
491    /// With `None` the arrow follows the buttons text color, `Some(color)` overrides it.
492    pub fn arrow_atom(color: Option<Color32>) -> Atom<'static> {
493        // A closure, so the size can be based on the `Ui`s spacing.
494        AtomKind::closure(move |ui, _args| {
495            let size = Vec2::splat(ui.spacing().icon_width);
496            IntoSizedResult {
497                intrinsic_size: size,
498                sized: SizedAtomKind::Paint {
499                    paint: Arc::new(move |ui: &Ui, args: AtomPaintArgs| {
500                        let color = color.unwrap_or(args.fallback_text_color);
501                        ui.painter().add(Self::arrow_shape(args.rect, color));
502                    }),
503                    size,
504                },
505            }
506        })
507        .into()
508    }
509
510    pub fn new(atoms: impl IntoAtoms<'a>) -> Self {
511        Self::from_button(Button::new(atoms.into_atoms()).right_text(Self::arrow_atom(None)))
512    }
513
514    /// Create a new submenu button from a [`Button`].
515    ///
516    /// Use [`Button::right_text`] and [`SubMenuButton::arrow_atom`] to add the default right
517    /// arrow.
518    pub fn from_button(button: Button<'a>) -> Self {
519        Self {
520            button,
521            sub_menu: SubMenu::default(),
522        }
523    }
524
525    /// Set the config for the submenu.
526    ///
527    /// The close behavior will not affect the current button, but the buttons in the submenu.
528    #[inline]
529    pub fn config(mut self, config: MenuConfig) -> Self {
530        self.sub_menu.config = Some(config);
531        self
532    }
533
534    /// Show the submenu button.
535    pub fn ui<R>(
536        self,
537        ui: &mut Ui,
538        content: impl FnOnce(&mut Ui) -> R,
539    ) -> (Response, Option<InnerResponse<R>>) {
540        let my_id = ui.next_auto_id();
541        let open = MenuState::from_ui(ui, |state, _| {
542            state.open_item == Some(SubMenu::id_from_widget_id(my_id))
543        });
544        let inactive = ui.style().visuals.widgets.inactive;
545        // TODO(lucasmerlin) add `open` function to `Button`
546        if open {
547            ui.style_mut().visuals.widgets.inactive = ui.style().visuals.widgets.open;
548        }
549        let response = self.button.ui(ui);
550        ui.style_mut().visuals.widgets.inactive = inactive;
551
552        let popup_response = self.sub_menu.show(ui, &response, content);
553
554        (response, popup_response)
555    }
556}

558/// Show a submenu in a menu.
559///
560/// Useful if you want to make custom menu buttons.
561/// Usually, just use [`MenuButton`] or [`SubMenuButton`] instead.
562#[derive(Clone, Debug, Default)]
563pub struct SubMenu {
564    config: Option<MenuConfig>,
565}
566
567impl SubMenu {
568    pub fn new() -> Self {
569        Self::default()
570    }
571
572    /// Set the config for the submenu.
573    ///
574    /// The close behavior will not affect the current button, but the buttons in the submenu.
575    #[inline]
576    pub fn config(mut self, config: MenuConfig) -> Self {
577        self.config = Some(config);
578        self
579    }
580
581    /// Get the id for the submenu from the widget/response id.
582    pub fn id_from_widget_id(widget_id: Id) -> Id {
583        widget_id.with("submenu")
584    }
585
586    /// Show the submenu.
587    ///
588    /// This does some heuristics to check if the `button_response` was the last thing in the
589    /// menu that was hovered/clicked, and if so, shows the submenu.
590    pub fn show<R>(
591        self,
592        ui: &Ui,
593        button_response: &Response,
594        content: impl FnOnce(&mut Ui) -> R,
595    ) -> Option<InnerResponse<R>> {
596        // This frame is only used to measure the offset we should set for the popup, so that
597        // contents align
598        let measurement_frame = Frame::menu(ui.style());
599
600        let id = Self::id_from_widget_id(button_response.id);
601
602        // Get the state from the parent menu
603        let pass_nr = ui.ctx().cumulative_pass_nr();
604        let (open_item, menu_id, parent_config, parent_arrow_focus) =
605            MenuState::from_ui(ui, |state, stack| {
606                state.last_visible_pass = pass_nr;
607                (
608                    state.open_item,
609                    stack.unique_id,
610                    MenuConfig::from_stack(stack),
611                    state.arrow_focus,
612                )
613            });
614
615        let mut menu_config = self.config.unwrap_or_else(|| parent_config.clone());
616        menu_config.bar = false;
617
618        #[expect(clippy::unwrap_used)] // Since we are a child of that ui, this should always exist
619        let menu_root_response = ui.ctx().read_response(menu_id).unwrap();
620
621        let hover_pos = ui.ctx().pointer_hover_pos();
622
623        // We don't care if the user is hovering over the border
624        let menu_rect = menu_root_response.rect - measurement_frame.total_margin();
625        let is_hovering_menu = hover_pos.is_some_and(|pos| {
626            ui.ctx().layer_id_at(pos) == Some(menu_root_response.layer_id)
627                && menu_rect.contains(pos)
628        });
629
630        let is_any_open = open_item.is_some();
631        let mut is_open = open_item == Some(id);
632        let was_open = is_open;
633        let mut set_open = None;
634
635        // We expand the button rect so there is no empty space where no menu is shown
636        // TODO(lucasmerlin): Instead, maybe make item_spacing.y 0.0?
637        let button_rect = button_response
638            .rect
639            .expand2(ui.style().spacing.item_spacing / 2.0);
640
641        // In theory some other widget could cover the button and this check would still pass
642        // But since we check if no other menu is open, nothing should be able to cover the button
643        let is_hovered = hover_pos.is_some_and(|pos| button_rect.contains(pos));
644
645        // `clicked` includes keyboard and accessibility click actions.
646        // Enter/Space on the focused button enters the submenu, an accessibility click toggles it,
647        // and pointer clicks keep it open (for touch and pointer interactions).
648        let clicked = button_response.clicked();
649        let clicked_by_pointer = button_response.clicked_by(PointerButton::Primary);
650        let clicked_by_keyboard_or_access = clicked && !clicked_by_pointer;
651        let focused = button_response.enabled() && button_response.has_focus();
652        let arrow_right =
653            focused && ui.memory(|mem| mem.focus_direction() == FocusDirection::Right);
654        if arrow_right {
655            ui.memory_mut(|mem| mem.move_focus(FocusDirection::None));
656        }
657        let enter = arrow_right || (focused && clicked_by_keyboard_or_access);
658        // Only arrow keys open on focus, so Tab can still move past the submenu.
659        let focus_open = focused && parent_arrow_focus == Some(button_response.id);
660        let keyboard_open = enter || focus_open || (!was_open && clicked_by_keyboard_or_access);
661        let pointer_changed =
662            ui.input(|i| i.pointer.delta() != Vec2::ZERO || i.pointer.any_click());
663        let (ignore_hover, left_submenu) = MenuState::from_id(ui.ctx(), id, |state| {
664            state.focus_first |= enter;
665            if keyboard_open {
666                // Keep the entry request through the popup's initial sizing pass.
667                state.last_visible_pass = pass_nr;
668                state.ignore_hover = true;
669            } else if !was_open || pointer_changed {
670                state.ignore_hover = false;
671            }
672            // Left inside the submenu moves focus back to our button.
673            let left_submenu = state.arrow_focus == Some(button_response.id);
674            if left_submenu {
675                state.arrow_focus = None;
676            }
677            (state.ignore_hover, left_submenu)
678        });
679
680        let focus_moved_away =
681            left_submenu || parent_arrow_focus.is_some_and(|focus| focus != button_response.id);
682        if is_open && (focus_moved_away || (clicked_by_keyboard_or_access && !enter)) {
683            set_open = Some(false);
684            is_open = false;
685        }
686
687        // The clicked handler is there for accessibility (keyboard navigation)
688        let should_open = button_response.enabled()
689            && (enter || focus_open || (!was_open && clicked) || (is_hovered && !is_any_open));
690        if should_open {
691            set_open = Some(true);
692            is_open = true;
693            // Ensure that all other sub menus are closed when we open the menu
694            MenuState::from_id(ui.ctx(), menu_id, |state| {
695                state.open_item = None;
696            });
697        }
698
699        // Half the margin brings the two menu frames edge-to-edge, then add the visual gap:
700        let gap = measurement_frame.total_margin().sum().x * 0.5 + menu_config.submenu_gap;
701
702        let mut response = button_response.clone();
703        // Expand the button rect so that the button and the first item in the submenu are aligned
704        let expand = Vec2::new(0.0, measurement_frame.total_margin().sum().y * 0.5);
705        response.interact_rect = response.interact_rect.expand2(expand);
706
707        let popup_response = Popup::from_response(&response)
708            .id(id)
709            .kind(PopupKind::Menu)
710            .open(is_open)
711            .align(RectAlign::RIGHT_START)
712            .layout(Layout::top_down_justified(Align::Min))
713            .gap(gap)
714            .style(menu_config.style.clone())
715            // The close behavior is handled by the menu (see below)
716            .close_behavior(PopupCloseBehavior::IgnoreClicks)
717            .info(
718                UiStackInfo::new(UiKind::Menu)
719                    .with_tag_value(MenuConfig::MENU_CONFIG_TAG, menu_config.clone()),
720            )
721            .show(|ui| {
722                // Ensure our layer stays on top when the button is clicked
723                if button_response.clicked() || button_response.is_pointer_button_down_on() {
724                    ui.ctx().move_to_top(ui.layer_id());
725                }
726                content(ui)
727            });
728
729        if let Some(popup_response) = &popup_response {
730            // If no child sub menu is open means we must be the deepest child sub menu.
731            let is_deepest_submenu = MenuState::is_deepest_open_sub_menu(ui.ctx(), id);
732
733            // If the user clicks and the cursor is not hovering over our menu rect, it's
734            // safe to assume they clicked outside the menu, so we close everything.
735            // If they were to hover some other parent submenu we wouldn't be open.
736            // Only edge case is the user hovering this submenu's button, so we also check
737            // if we clicked outside the parent menu (which we luckily have access to here).
738            let clicked_outside = is_deepest_submenu
739                && popup_response.response.clicked_elsewhere()
740                && menu_root_response.clicked_elsewhere();
741
742            // We never automatically close when a submenu button is clicked, (so menus work
743            // on touch devices)
744            // Luckily we will always be the deepest submenu when a submenu button is clicked,
745            // so the following check is enough.
746            let submenu_button_clicked = button_response.clicked();
747
748            let clicked_inside = is_deepest_submenu
749                && !submenu_button_clicked
750                && response.ctx.input(|i| i.pointer.any_click())
751                && hover_pos.is_some_and(|pos| popup_response.response.interact_rect.contains(pos));
752
753            let click_close = match menu_config.close_behavior {
754                PopupCloseBehavior::CloseOnClick => clicked_outside || clicked_inside,
755                PopupCloseBehavior::CloseOnClickOutside => clicked_outside,
756                PopupCloseBehavior::IgnoreClicks => false,
757            };
758
759            if click_close {
760                set_open = Some(false);
761                ui.close();
762            }
763
764            let is_moving_towards_rect = ui.input(|i| {
765                i.pointer
766                    .is_moving_towards_rect(&popup_response.response.rect)
767            });
768            if is_moving_towards_rect {
769                // We need to repaint while this is true, so we can detect when
770                // the pointer is no longer moving towards the rect
771                ui.request_repaint();
772            }
773            let hovering_other_menu_entry = is_open
774                && !ignore_hover
775                && !is_hovered
776                && !popup_response.response.contains_pointer()
777                && !is_moving_towards_rect
778                && is_hovering_menu;
779
780            let close_called = popup_response.response.should_close();
781
782            // Close the parent ui to e.g. close the popup from where the submenu was opened
783            if close_called {
784                ui.close();
785            }
786
787            if hovering_other_menu_entry {
788                set_open = Some(false);
789            }
790
791            if ui.will_parent_close() {
792                ui.data_mut(|data| data.remove_by_type::<MenuState>());
793            }
794        }
795
796        if let Some(set_open) = set_open {
797            MenuState::from_id(ui.ctx(), menu_id, |state| {
798                state.open_item = set_open.then_some(id);
799            });
800        }
801
802        popup_response
803    }
804}
