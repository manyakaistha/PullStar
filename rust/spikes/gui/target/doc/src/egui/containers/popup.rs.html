12/// What should we anchor the popup to?
13///
14/// The final position for the popup will be calculated based on [`RectAlign`]
15/// and can be customized with [`Popup::align`] and [`Popup::align_alternatives`].
16/// [`PopupAnchor`] is the parent rect of [`RectAlign`].
17///
18/// For [`PopupAnchor::Pointer`], [`PopupAnchor::PointerFixed`] and [`PopupAnchor::Position`],
19/// the rect will be derived via [`Rect::from_pos`] (so a zero-sized rect at the given position).
20///
21/// The rect should be in global coordinates. `PopupAnchor::from(&response)` will automatically
22/// do this conversion.
23#[derive(Clone, Copy, Debug, PartialEq, Eq)]
24pub enum PopupAnchor {
25    /// Show the popup relative to some parent [`Rect`].
26    ParentRect(Rect),
27
28    /// Show the popup relative to the mouse pointer.
29    Pointer,
30
31    /// Remember the mouse position and show the popup relative to that (like a context menu).
32    PointerFixed,
33
34    /// Show the popup relative to a specific position.
35    Position(Pos2),
36}
37
38impl From<Rect> for PopupAnchor {
39    fn from(rect: Rect) -> Self {
40        Self::ParentRect(rect)
41    }
42}
43
44impl From<Pos2> for PopupAnchor {
45    fn from(pos: Pos2) -> Self {
46        Self::Position(pos)
47    }
48}
49
50impl From<&Response> for PopupAnchor {
51    fn from(response: &Response) -> Self {
52        // We use interact_rect so we don't show the popup relative to some clipped point
53        let mut widget_rect = response.interact_rect;
54        if let Some(to_global) = response.ctx.layer_transform_to_global(response.layer_id) {
55            widget_rect = to_global * widget_rect;
56        }
57        Self::ParentRect(widget_rect)
58    }
59}
60
61impl PopupAnchor {
62    /// Get the rect the popup should be shown relative to.
63    /// Returns `Rect::from_pos` for [`PopupAnchor::Pointer`], [`PopupAnchor::PointerFixed`]
64    /// and [`PopupAnchor::Position`] (so the rect will be zero-sized).
65    pub fn rect(self, popup_id: Id, ctx: &Context) -> Option<Rect> {
66        match self {
67            Self::ParentRect(rect) => Some(rect),
68            Self::Pointer => ctx.pointer_hover_pos().map(Rect::from_pos),
69            Self::PointerFixed => Popup::position_of_id(ctx, popup_id).map(Rect::from_pos),
70            Self::Position(pos) => Some(Rect::from_pos(pos)),
71        }
72    }
73}

75/// Determines popup's close behavior
76#[derive(Clone, Copy, PartialEq, Eq, Default, Debug)]
77pub enum PopupCloseBehavior {
78    /// Popup will be closed on click anywhere, inside or outside the popup.
79    ///
80    /// It is used in [`crate::ComboBox`] and in [`crate::containers::menu`]s.
81    #[default]
82    CloseOnClick,
83
84    /// Popup will be closed if the click happened somewhere else
85    /// but in the popup's body
86    CloseOnClickOutside,
87
88    /// Clicks will be ignored. Popup might be closed manually by calling [`Popup::close_all`]
89    /// or by pressing the escape button
90    IgnoreClicks,
91}

108/// How do we determine if the popup should be open or closed
109enum OpenKind<'a> {
110    /// Always open
111    Open,
112
113    /// Always closed
114    Closed,
115
116    /// Open if the bool is true
117    Bool(&'a mut bool),
118
119    /// Store the open state via [`crate::Memory`]
120    Memory { set: Option<SetOpenCommand> },
121}
122
123impl OpenKind<'_> {
124    /// Returns `true` if the popup should be open
125    fn is_open(&self, popup_id: Id, ctx: &Context) -> bool {
126        match self {
127            OpenKind::Open => true,
128            OpenKind::Closed => false,
129            OpenKind::Bool(open) => **open,
130            OpenKind::Memory { .. } => Popup::is_id_open(ctx, popup_id),
131        }
132    }
133}

143impl PopupKind {
144    /// Returns the order to be used with this kind.
145    pub fn order(self) -> Order {
146        match self {
147            Self::Tooltip => Order::Tooltip,
148            Self::Menu | Self::Popup => Order::Foreground,
149        }
150    }

512    /// Calculate the best alignment for the popup, based on the last size and screen rect.
513    pub fn get_best_align(&self) -> RectAlign {
514        let expected_popup_size = self
515            .get_expected_size()
516            .unwrap_or_else(|| vec2(self.width.unwrap_or(0.0), 0.0));
517
518        let Some(anchor_rect) = self.anchor.rect(self.id, &self.ctx) else {
519            return self.rect_align;
520        };
521
522        RectAlign::find_best_align(
523            #[expect(clippy::iter_on_empty_collections)]
524            #[expect(clippy::or_fun_call)]
525            core::iter::chain(
526                once(self.rect_align),
527                self.alternative_aligns
528                    // Need the empty slice so the iters have the same type so we can unwrap_or
529                    .map(|a| core::iter::chain(a.iter().copied(), [].iter().copied()))
530                    .unwrap_or(core::iter::chain(
531                        self.rect_align.symmetries().iter().copied(),
532                        RectAlign::MENU_ALIGNS.iter().copied(),
533                    )),
534            ),
535            self.ctx.content_rect(),
536            anchor_rect,
537            self.gap,
538            expected_popup_size,
539        )
540        .unwrap_or_default()
541    }

543    /// Show the popup.
544    ///
545    /// Returns `None` if the popup is not open or anchor is `PopupAnchor::Pointer` and there is
546    /// no pointer.
547    pub fn show<R>(self, content: impl FnOnce(&mut Ui) -> R) -> Option<InnerResponse<R>> {
548        let id = self.id;
549        // When the popup was just opened with a click we don't want to immediately close it based
550        // on the `PopupCloseBehavior`, so we need to remember if the popup was already open on
551        // last frame. A convenient way to check this is to see if we have a response for the `Area`
552        // from last frame:
553        let was_open_last_frame = self.ctx.read_response(id).is_some();
554
555        let hover_pos = self.ctx.pointer_hover_pos();
556        if let OpenKind::Memory { set } = self.open_kind {
557            match set {
558                Some(SetOpenCommand::Bool(open)) => {
559                    if open {
560                        match self.anchor {
561                            PopupAnchor::PointerFixed => {
562                                self.ctx.memory_mut(|mem| mem.open_popup_at(id, hover_pos));
563                            }
564                            _ => Popup::open_id(&self.ctx, id),
565                        }
566                    } else {
567                        Self::close_id(&self.ctx, id);
568                    }
569                }
570                Some(SetOpenCommand::Toggle) => {
571                    Self::toggle_id(&self.ctx, id);
572                }
573                None => {
574                    self.ctx.memory_mut(|mem| mem.keep_popup_open(id));
575                }
576            }
577        }
578
579        if !self.open_kind.is_open(self.id, &self.ctx) {
580            return None;
581        }
582
583        let best_align = self.get_best_align();
584
585        let Popup {
586            id,
587            ctx,
588            anchor,
589            open_kind,
590            close_behavior,
591            kind,
592            info,
593            layer_id,
594            rect_align: _,
595            alternative_aligns: _,
596            gap,
597            width,
598            sizing_pass,
599            sense,
600            interactable,
601            layout,
602            frame,
603            style,
604            anchor_widget,
605            accessibility_label,
606        } = self;
607
608        if kind != PopupKind::Tooltip {
609            ctx.pass_state_mut(|fs| {
610                fs.layers
611                    .entry(layer_id)
612                    .or_default()
613                    .open_popups
614                    .insert(id)
615            });
616        }
617
618        let anchor_rect = anchor.rect(id, &ctx)?;
619
620        let (pivot, anchor) = best_align.pivot_pos(&anchor_rect, gap);
621
622        let mut area = Area::new(id)
623            .order(kind.order())
624            .pivot(pivot)
625            .fixed_pos(anchor)
626            .sense(sense)
627            .interactable(interactable)
628            .layout(layout)
629            .sizing_pass(sizing_pass || !was_open_last_frame)
630            .info(info.unwrap_or_else(|| {
631                UiStackInfo::new(kind.into()).with_tag_value(
632                    MenuConfig::MENU_CONFIG_TAG,
633                    MenuConfig::new()
634                        .close_behavior(close_behavior)
635                        .style(style.clone()),
636                )
637            }));
638
639        if let Some(width) = width {
640            area = area.default_width(width);
641        }
642        if let Some(anchor_widget) = anchor_widget {
643            area = area.accessibility_parent(anchor_widget);
644        }
645        if let Some(label) = accessibility_label {
646            area = area.accessible_name(label);
647        }
648
649        let mut response = area.show(&ctx, |ui| {
650            style.apply(ui.style_mut());
651            let frame = frame.unwrap_or_else(|| Frame::popup(ui.style()));
652            let inner = frame.show(ui, content).inner;
653            if kind == PopupKind::Menu {
654                MenuState::handle_keyboard(ui, anchor_widget);
655            }
656            inner
657        });
658
659        // Only a click whose press started while the popup was already open may close it.
660        // Otherwise a popup opened on a pointer *press* (e.g. a right-press context menu)
661        // would immediately close again on the following *release*.
662        // This also covers the case where the popup was just opened with a click.
663        let press_started_while_open_id = id.with("press_started_while_open");
664        let (any_pressed, any_click) =
665            ctx.input(|i| (i.pointer.any_pressed(), i.pointer.any_click()));
666        let press_started_while_open = if !was_open_last_frame {
667            ctx.data_mut(|d| d.remove::<bool>(press_started_while_open_id));
668            false
669        } else if any_pressed {
670            ctx.data_mut(|d| d.insert_temp(press_started_while_open_id, true));
671            true
672        } else {
673            ctx.data(|d| d.get_temp(press_started_while_open_id))
674                .unwrap_or(false)
675        };
676        let close_click = press_started_while_open && any_click;
677
678        let closed_by_click = match close_behavior {
679            PopupCloseBehavior::CloseOnClick => close_click,
680            PopupCloseBehavior::CloseOnClickOutside => {
681                close_click && response.response.clicked_elsewhere()
682            }
683            PopupCloseBehavior::IgnoreClicks => false,
684        };
685
686        // Mark the menu as shown, so the sub menu open state is not reset
687        MenuState::mark_shown(&ctx, id);
688
689        // If a submenu is open, the CloseBehavior is handled there
690        let is_any_submenu_open = !MenuState::is_deepest_open_sub_menu(&response.response.ctx, id);
691
692        let should_close = (!is_any_submenu_open && closed_by_click)
693            || ctx.input(|i| i.key_pressed(Key::Escape))
694            || response.response.should_close();
695
696        if should_close {
697            response.response.set_close();
698            ctx.data_mut(|d| d.remove::<bool>(press_started_while_open_id));
699        }
700
701        match open_kind {
702            OpenKind::Open | OpenKind::Closed => {}
703            OpenKind::Bool(open) => {
704                if should_close {
705                    *open = false;
706                }
707            }
708            OpenKind::Memory { .. } => {
709                if should_close {
710                    ctx.memory_mut(|mem| mem.close_popup(id));
711                }
712            }
713        }
714
715        Some(response)
716    }
717}

721    /// The default ID when constructing a popup from the [`Response`] of e.g. a button.
722    pub fn default_response_id(response: &Response) -> Id {
723        response.id.with("popup")
724    }
725
726    /// Is the given popup open?
727    ///
728    /// This assumes the use of either:
729    /// * [`Self::open_memory`]
730    /// * [`Self::from_toggle_button_response`]
731    /// * [`Self::menu`]
732    /// * [`Self::context_menu`]
733    ///
734    /// The popup id should be the same as either you set with [`Self::id`] or the
735    /// default one from [`Self::default_response_id`].
736    pub fn is_id_open(ctx: &Context, popup_id: Id) -> bool {
737        ctx.memory(|mem| mem.is_popup_open(popup_id))
738    }
739
740    /// Is any popup open?
741    ///
742    /// This assumes the egui memory is being used to track the open state of popups.
743    pub fn is_any_open(ctx: &Context) -> bool {
744        ctx.memory(|mem| mem.any_popup_open())
745    }
746
747    /// Open the given popup and close all others.
748    pub fn open_id(ctx: &Context, popup_id: Id) {
749        ctx.memory_mut(|mem| mem.open_popup(popup_id));
750    }
751
752    /// Toggle the given popup between closed and open.
753    ///
754    /// Note: At most, only one popup can be open at a time.
755    pub fn toggle_id(ctx: &Context, popup_id: Id) {
756        ctx.memory_mut(|mem| mem.toggle_popup(popup_id));
757    }
758
759    /// Close all currently open popups.
760    pub fn close_all(ctx: &Context) {
761        ctx.memory_mut(|mem| mem.close_all_popups());
762    }
763
764    /// Close the given popup, if it is open.
765    ///
766    /// See also [`Self::close_all`] if you want to close any / all currently open popups.
767    pub fn close_id(ctx: &Context, popup_id: Id) {
768        ctx.memory_mut(|mem| mem.close_popup(popup_id));
769    }
770
771    /// Get the position for this popup, if it is open.
772    pub fn position_of_id(ctx: &Context, popup_id: Id) -> Option<Pos2> {
773        ctx.memory(|mem| mem.popup_position(popup_id))
774    }
775}
