5/// Align some widgets of unknown size within the available space of a [`Ui`].
6///
7/// For instance, this can put a group of widgets in the bottom-right corner of the parent [`Ui`].
8///
9/// egui is an immediate mode GUI, so the size of the contents is not known until they have been added.
10/// This container remembers the size from the previous pass, and uses that to place the contents.
11/// On the first pass (when no size is known yet) the contents are added invisibly,
12/// and a new pass is requested with [`crate::Context::request_discard`], so there is no visible flicker.
13///
14/// Like a [`crate::Panel`], this reserves space in the parent [`Ui`] along its main axis,
15/// so you can e.g. put something at the bottom of a vertical [`Ui`] and keep adding widgets above it:
16///
17/// ```
18/// # egui::__run_test_ui(|ui| {
19/// egui::Aligned::new(egui::Align2::RIGHT_BOTTOM).show(ui, |ui| {
20///     let _ = ui.button("Bottom right");
21/// });
22/// ui.label("This goes at the top, in the space left above the button");
23/// # });
24/// ```
25///
26/// More precisely: if the contents are aligned to the _end_ of the parent's main axis
27/// (e.g. [`Align::Max`] in a top-down [`Ui`]), the cursor is moved so that widgets added afterwards
28/// end up _before_ the contents. Otherwise the cursor advances past the contents, as usual.
29///
30/// ```
31/// # egui::__run_test_ui(|ui| {
32/// egui::Aligned::new(egui::Align2::RIGHT_BOTTOM).show(ui, |ui| {
33///     ui.label("Bottom right");
34///     let _ = ui.button("Click me");
35/// });
36/// # });
37/// ```
38#[must_use = "You should call .show()"]
39#[derive(Clone, Copy, Debug)]
40pub struct Aligned {
41    align2: Align2,
42    id_salt: Option<IdSalt>,
43}
44
45impl Aligned {
46    /// Align the contents within the available space of the parent [`Ui`].
47    pub fn new(align2: Align2) -> Self {
48        Self {
49            align2,
50            id_salt: None,
51        }
52    }
53
54    /// A source for the unique [`crate::Id`], e.g. `.id_salt("second_aligned")` or `.id_salt(loop_index)`.
55    ///
56    /// Needed if you have several [`Aligned`] in the same [`Ui`].
57    #[inline]
58    pub fn id_salt(mut self, id_salt: impl AsIdSalt) -> Self {
59        self.id_salt = Some(IdSalt::new(id_salt));
60        self
61    }
62
63    /// Show the contents.
64    pub fn show<R>(self, ui: &mut Ui, add_contents: impl FnOnce(&mut Ui) -> R) -> InnerResponse<R> {
65        let Self { align2, id_salt } = self;
66
67        let id_salt = id_salt.unwrap_or_else(|| IdSalt::new("aligned"));
68        let id = ui.make_persistent_id(id_salt);
69        let size_id = id.with("size");
70
71        let available_rect = ui.available_rect_before_wrap();
72        let cursor_before = ui.cursor();
73        let last_size: Option<Vec2> = ui.data(|d| d.get_temp(size_id));
74
75        let content_rect = if let Some(size) = last_size {
76            let left_top = place(align2, size, available_rect).left_top();
77
78            // Keep the full available size, in case the contents grow this pass:
79            Rect::from_min_size(left_top, available_rect.size())
80        } else {
81            available_rect
82        };
83
84        let mut builder = UiBuilder::new().scope_id(id).max_rect(content_rect);
85        if last_size.is_none() {
86            // We don't know the size yet, so we can't place the contents correctly.
87            // Add them invisibly to measure them, and then try again.
88            builder = builder.invisible();
89        }
90
91        let response = ui.scope_builder(builder, add_contents);
92
93        let size = response.response.rect.size();
94        if last_size != Some(size) {
95            ui.ctx().request_discard("Aligned: size changed");
96        }
97        ui.data_mut(|d| d.insert_temp(size_id, size));
98
99        // Reserve space in the parent like a `Panel` would:
100        // if the contents sit at the end of the parent's main axis,
101        // move the cursor so that further widgets are placed _before_ the contents.
102        // (`scope_builder` has already advanced the cursor _past_ the contents, which is
103        // the right thing when they are at the start of the main axis, or centered.)
104        //
105        // We use where the contents _will_ end up (given their now known size),
106        // so that this is correct also on the invisible measuring pass.
107        let placed_rect = place(align2, size, available_rect);
108        let main_dir = ui.layout().main_dir();
109        let axis = usize::from(main_dir.is_vertical()); // 0 = x, 1 = y
110        let spacing = ui.spacing().item_spacing[axis];
111        let mut cursor = cursor_before;
112        match (main_dir, align2.0[axis]) {
113            (Direction::LeftToRight | Direction::TopDown, Align::Max) => {
114                cursor.max[axis] = placed_rect.min[axis] - spacing;
115                ui.set_cursor(cursor);
116            }
117            (Direction::RightToLeft | Direction::BottomUp, Align::Min) => {
118                cursor.min[axis] = placed_rect.max[axis] + spacing;
119                ui.set_cursor(cursor);
120            }
121            _ => {}
122        }
123
124        response
125    }
126}

128/// Where to put contents of the given size, aligned within `available_rect`.
129///
130/// Never places the contents above or to the left of the available space,
131/// even if the contents are larger than the available space.
132fn place(align2: Align2, size: Vec2, available_rect: Rect) -> Rect {
133    let left_top = align2
134        .align_size_within_rect(size, available_rect)
135        .left_top()
136        .max(available_rect.left_top());
137    Rect::from_min_size(left_top, size)
138}

140#[cfg(test)]
141mod tests {
142    use emath::vec2;
143
144    use crate::{Align2, Aligned, Context, FontDefinitions, Rect};
145
146    /// Run one frame of a single [`Aligned`] and return the rect of its contents, and the rect of the parent.
147    fn run_frame(ctx: &Context) -> (Rect, Rect) {
148        let mut rects = None;
149        let output = ctx.run_ui(Default::default(), |ui| {
150            let available_rect = ui.available_rect_before_wrap();
151            let response = Aligned::new(Align2::RIGHT_BOTTOM).show(ui, |ui| {
152                ui.allocate_space(vec2(30.0, 20.0));
153            });
154            rects = Some((response.response.rect, available_rect));
155        });
156        output.drop_without_applying_deltas();
157        rects.expect("The closure was not called")
158    }
159
160    #[test]
161    fn contents_end_up_aligned() {
162        let ctx = Context::default();
163        ctx.set_fonts(FontDefinitions::empty());
164
165        // First frame measures, second frame places:
166        run_frame(&ctx);
167        let (content_rect, available_rect) = run_frame(&ctx);
168
169        assert_eq!(content_rect.size(), vec2(30.0, 20.0));
170        assert!((content_rect.right() - available_rect.right()).abs() < 0.01);
171        assert!((content_rect.bottom() - available_rect.bottom()).abs() < 0.01);
172    }
