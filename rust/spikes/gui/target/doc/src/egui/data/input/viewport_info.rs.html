3/// An input event from the backend into egui, about a specific [viewport](crate::viewport).
4#[derive(Clone, Copy, Debug, PartialEq, Eq)]
5#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
6pub enum ViewportEvent {
7    /// The user clicked the close-button on the window, or similar.
8    ///
9    /// If this is the root viewport, the application will exit
10    /// after this frame unless you send a
11    /// [`crate::ViewportCommand::CancelClose`] command.
12    ///
13    /// If this is not the root viewport,
14    /// it is up to the user to hide this viewport the next frame.
15    ///
16    /// This even will wake up both the child and parent viewport.
17    Close,
18}
19
20/// Information about the current viewport, given as input each frame.
21///
22/// `None` means "unknown".
23///
24/// All units are in ui "points", which can be calculated from native physical pixels
25/// using `pixels_per_point` = [`crate::Context::zoom_factor`] * `[Self::native_pixels_per_point`];
26#[derive(Clone, Debug, Default, PartialEq)]
27#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
28pub struct ViewportInfo {
29    /// Parent viewport, if known.
30    pub parent: Option<crate::ViewportId>,
31
32    /// Name of the viewport, if known.
33    pub title: Option<String>,
34
35    pub events: Vec<ViewportEvent>,
36
37    /// The OS native pixels-per-point.
38    ///
39    /// This should always be set, if known.
40    ///
41    /// On web this takes browser scaling into account,
42    /// and corresponds to [`window.devicePixelRatio`](https://developer.mozilla.org/en-US/docs/Web/API/Window/devicePixelRatio) in JavaScript.
43    pub native_pixels_per_point: Option<f32>,
44
45    /// Current monitor size in egui points.
46    pub monitor_size: Option<Vec2>,
47
48    /// The inner rectangle of the native window, in monitor space and ui points scale.
49    ///
50    /// This is the content rectangle of the viewport.
51    ///
52    /// **`eframe` notes**:
53    ///
54    /// On Android / Wayland, this will always be `None` since getting the
55    /// position of the window is not possible.
56    pub inner_rect: Option<Rect>,
57
58    /// The outer rectangle of the native window, in monitor space and ui points scale.
59    ///
60    /// This is the content rectangle plus decoration chrome.
61    ///
62    /// **`eframe` notes**:
63    ///
64    /// On Android / Wayland, this will always be `None` since getting the
65    /// position of the window is not possible.
66    pub outer_rect: Option<Rect>,
67
68    /// Are we minimized?
69    pub minimized: Option<bool>,
70
71    /// Are we maximized?
72    pub maximized: Option<bool>,
73
74    /// Are we in fullscreen mode?
75    pub fullscreen: Option<bool>,
76
77    /// Is the window focused and able to receive input?
78    ///
79    /// This should be the same as [`RawInput::focused`](crate::RawInput::focused).
80    pub focused: Option<bool>,
81
82    /// Is the window fully occluded (completely covered) by another window?
83    ///
84    /// Not all platforms support this.
85    /// On platforms that don't, this will be `None` or `Some(false)`.
86    pub occluded: Option<bool>,
87}
88
89impl ViewportInfo {
90    /// Is the window considered visible for rendering purposes?
91    ///
92    /// A window is not visible if it is minimized or occluded.
93    /// When not visible, the UI is not painted and rendering is skipped,
94    /// but application logic may still be executed by some integrations.
95    ///
96    /// Send [`crate::ViewportCommand::RequestPaintWhileHidden`] to have the ui run and the
97    /// frame painted anyway, e.g. to screenshot or drive an app that sits in the background.
98    pub fn visible(&self) -> Option<bool> {
99        match (self.minimized, self.occluded) {
100            (Some(true), _) | (_, Some(true)) => Some(false),
101            (Some(false), Some(false)) => Some(true),
102            (_, None) | (None, _) => None,
103        }
104    }
105
106    /// This viewport has been told to close.
107    ///
108    /// If this is the root viewport, the application will exit
109    /// after this frame unless you send a
110    /// [`crate::ViewportCommand::CancelClose`] command.
111    ///
112    /// If this is not the root viewport,
113    /// it is up to the user to hide this viewport the next frame.
114    pub fn close_requested(&self) -> bool {
115        self.events.contains(&ViewportEvent::Close)
116    }
117
118    /// Helper: move [`Self::events`], clone the other fields.
119    pub fn take(&mut self) -> Self {
120        Self {
121            parent: self.parent,
122            title: self.title.clone(),
123            events: core::mem::take(&mut self.events),
124            native_pixels_per_point: self.native_pixels_per_point,
125            monitor_size: self.monitor_size,
126            inner_rect: self.inner_rect,
127            outer_rect: self.outer_rect,
128            minimized: self.minimized,
129            maximized: self.maximized,
130            fullscreen: self.fullscreen,
131            focused: self.focused,
132            occluded: self.occluded,
133        }
134    }
135
136    pub fn ui(&self, ui: &mut crate::Ui) {
137        let Self {
138            parent,
139            title,
140            events,
141            native_pixels_per_point,
142            monitor_size,
143            inner_rect,
144            outer_rect,
145            minimized,
146            maximized,
147            fullscreen,
148            focused,
149            occluded,
150        } = self;
151
152        crate::Grid::new("viewport_info").show(ui, |ui| {
153            ui.label("Parent:");
154            ui.label(opt_as_str(parent));
155            ui.end_row();
156
157            ui.label("Title:");
158            ui.label(opt_as_str(title));
159            ui.end_row();
160
161            ui.label("Events:");
162            ui.label(format!("{events:?}"));
163            ui.end_row();
164
165            ui.label("Native pixels-per-point:");
166            ui.label(opt_as_str(native_pixels_per_point));
167            ui.end_row();
168
169            ui.label("Monitor size:");
170            ui.label(opt_as_str(monitor_size));
171            ui.end_row();
172
173            ui.label("Inner rect:");
174            ui.label(opt_rect_as_string(inner_rect));
175            ui.end_row();
176
177            ui.label("Outer rect:");
178            ui.label(opt_rect_as_string(outer_rect));
179            ui.end_row();
180
181            ui.label("Minimized:");
182            ui.label(opt_as_str(minimized));
183            ui.end_row();
184
185            ui.label("Maximized:");
186            ui.label(opt_as_str(maximized));
187            ui.end_row();
188
189            ui.label("Fullscreen:");
190            ui.label(opt_as_str(fullscreen));
191            ui.end_row();
192
193            ui.label("Focused:");
194            ui.label(opt_as_str(focused));
195            ui.end_row();
196
197            ui.label("Occluded:");
198            ui.label(opt_as_str(occluded));
199            ui.end_row();
200
201            let visible = self.visible();
202
203            ui.label("Visible:");
204            ui.label(opt_as_str(&visible));
205            ui.end_row();
206
207            #[expect(clippy::ref_option)]
208            fn opt_rect_as_string(v: &Option<Rect>) -> String {
209                v.as_ref().map_or(String::new(), |r| {
210                    format!("Pos: {:?}, size: {:?}", r.min, r.size())
211                })
212            }
213
214            #[expect(clippy::ref_option)]
215            fn opt_as_str<T: core::fmt::Debug>(v: &Option<T>) -> String {
216                v.as_ref().map_or(String::new(), |v| format!("{v:?}"))
217            }
218        });
219    }
