13/// An input event generated by the integration.
14///
15/// This only covers events that egui cares about.
16#[derive(Clone, Debug, PartialEq)]
17#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
18pub enum Event {
19    /// The integration detected a "copy" event (e.g. Cmd+C).
20    Copy,
21
22    /// The integration detected a "cut" event (e.g. Cmd+X).
23    Cut,
24
25    /// The integration detected a "paste" event (e.g. Cmd+V).
26    Paste(String),
27
28    /// The integration detected a "paste" event (e.g. Cmd+V) where the clipboard held an
29    /// image instead of text (e.g. a screenshot, or an image copied from another app).
30    ///
31    /// Mirrors [`crate::OutputCommand::CopyImage`] for the opposite direction: an integration
32    /// that supports copying an image out (via `arboard`, say) should support pasting one back
33    /// in the same way. Only emitted when the clipboard has no usable text representation —
34    /// [`Self::Paste`] still takes priority when both are available.
35    PasteImage(alloc::sync::Arc<ColorImage>),
36
37    /// Paste the X11/Wayland PRIMARY selection where the user middle-clicked.
38    ///
39    /// Unlike [`Self::Paste`], this goes to the hovered `TextEdit` and inserts the text at `pos`.
40    MiddleClickPaste {
41        /// Where the user clicked, in points.
42        pos: Pos2,
43
44        /// The contents of the PRIMARY selection.
45        text: String,
46    },
47
48    /// Text input, e.g. via keyboard.
49    ///
50    /// When the user presses enter/return, do not send a [`Text`](Event::Text) (just [`Key::Enter`]).
51    Text(String),
52
53    /// A key was pressed or released.
54    ///
55    /// ## Note for integration authors
56    ///
57    /// Key events that has been processed by IMEs should not be sent to `egui`.
58    Key {
59        /// Most of the time, it's the logical key, heeding the active keymap -- for instance, if the user has Dvorak
60        /// keyboard layout, it will be taken into account.
61        ///
62        /// If it's impossible to determine the logical key on desktop platforms (say, in case of non-Latin letters),
63        /// `key` falls back to the value of the corresponding physical key. This is necessary for proper work of
64        /// standard shortcuts that only respond to Latin-based bindings (such as `Ctrl` + `V`).
65        key: Key,
66
67        /// The physical key, corresponding to the actual position on the keyboard.
68        ///
69        /// This ignores keymaps, so it is not recommended to use this.
70        /// The only thing it makes sense for is things like games,
71        /// where e.g. the physical location of WSAD on QWERTY should always map to movement,
72        /// even if the user is using Dvorak or AZERTY.
73        ///
74        /// `eframe` does not (yet) implement this on web.
75        physical_key: Option<Key>,
76
77        /// Was it pressed or released?
78        pressed: bool,
79
80        /// If this is a `pressed` event, is it a key-repeat?
81        ///
82        /// On many platforms, holding down a key produces many repeated "pressed" events for it, so called key-repeats.
83        /// Sometimes you will want to ignore such events, and this lets you do that.
84        ///
85        /// egui will automatically detect such repeat events and mark them as such here.
86        /// Therefore, if you are writing an egui integration, you do not need to set this (just set it to `false`).
87        repeat: bool,
88
89        /// The state of the modifier keys at the time of the event.
90        modifiers: Modifiers,
91    },
92
93    /// The set of held modifier keys changed.
94    ModifiersChanged(Modifiers),
95
96    /// The mouse or touch moved to a new place.
97    PointerMoved(Pos2),
98
99    /// The mouse moved, the units are unspecified.
100    /// Represents the actual movement of the mouse, without acceleration or clamped by screen edges.
101    /// `PointerMoved` and `MouseMoved` can be sent at the same time.
102    /// This event is optional. If the integration can not determine unfiltered motion it should not send this event.
103    MouseMoved(Vec2),
104
105    /// A mouse button was pressed or released (or a touch started or stopped).
106    PointerButton {
107        /// Where is the pointer?
108        pos: Pos2,
109
110        /// What mouse button? For touches, use [`PointerButton::Primary`].
111        button: PointerButton,
112
113        /// Was it the button/touch pressed this frame, or released?
114        pressed: bool,
115
116        /// The state of the modifier keys at the time of the event.
117        modifiers: Modifiers,
118    },
119
120    /// The mouse left the screen, or the last/primary touch input disappeared.
121    ///
122    /// This means there is no longer a cursor on the screen for hovering etc.
123    ///
124    /// On touch-up first send `PointerButton{pressed: false, …}` followed by `PointerLeft`.
125    PointerGone,
126
127    /// Zoom scale factor this frame (e.g. from a pinch gesture).
128    ///
129    /// * `zoom = 1`: no change.
130    /// * `zoom < 1`: pinch together
131    /// * `zoom > 1`: pinch spread
132    ///
133    /// Note that egui also implement zooming by holding `Ctrl` and scrolling the mouse wheel,
134    /// so integration need NOT emit this `Zoom` event in those cases, just [`Self::MouseWheel`].
135    ///
136    /// As a user, check [`crate::InputState::smooth_scroll_delta`] to see if the user did any zooming this frame.
137    Zoom(f32),
138
139    /// Rotation in radians this frame, measuring clockwise (e.g. from a rotation gesture).
140    Rotate(f32),
141
142    /// IME Event
143    Ime(ImeEvent),
144
145    /// On touch screens, report this *in addition to*
146    /// [`Self::PointerMoved`], [`Self::PointerButton`], [`Self::PointerGone`]
147    Touch {
148        /// Hashed device identifier (if available; may be zero).
149        /// Can be used to separate touches from different devices.
150        device_id: TouchDeviceId,
151
152        /// Unique identifier of a finger/pen. Value is stable from touch down
153        /// to lift-up
154        id: TouchId,
155
156        /// One of: start move end cancel.
157        phase: TouchPhase,
158
159        /// Position of the touch (or where the touch was last detected)
160        pos: Pos2,
161
162        /// Describes how hard the touch device was pressed. May always be `None` if the platform does
163        /// not support pressure sensitivity.
164        /// The value is in the range from 0.0 (no pressure) to 1.0 (maximum pressure).
165        force: Option<f32>,
166    },
167
168    /// A raw mouse wheel event as sent by the backend.
169    ///
170    /// Used for scrolling.
171    MouseWheel {
172        /// The unit of `delta`: points, lines, or pages.
173        unit: MouseWheelUnit,
174
175        /// The direction of the vector indicates how to move the _content_ that is being viewed.
176        /// So if you get positive values, the content being viewed should move to the right and down,
177        /// revealing new things to the left and up.
178        ///
179        /// A positive X-value indicates the content is being moved right,
180        /// as when swiping right on a touch-screen or track-pad with natural scrolling.
181        ///
182        /// A positive Y-value indicates the content is being moved down,
183        /// as when swiping down on a touch-screen or track-pad with natural scrolling.
184        delta: Vec2,
185
186        /// The phase of the scroll, useful for trackpads.
187        ///
188        /// If unknown set this to [`TouchPhase::Move`].
189        phase: TouchPhase,
190
191        /// What is driving the scroll: a wheel, fingers on a trackpad,
192        /// or the system continuing a finger scroll with momentum.
193        ///
194        /// Set this to [`MouseWheelSource::Unknown`] if unknown.
195        source: MouseWheelSource,
196
197        /// The state of the modifier keys at the time of the event.
198        modifiers: Modifiers,
199    },
200
201    /// The native window gained or lost focused (e.g. the user clicked alt-tab).
202    WindowFocused(bool),
203
204    /// An assistive technology (e.g. screen reader) requested an action.
205    AccessKitActionRequest(accesskit::ActionRequest),
206}
