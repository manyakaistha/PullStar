869fn test_key_from_name() {
870    assert_eq!(
871        Key::ALL.len(),
872        Key::Clear as usize + 1,
873        "Some keys are missing in Key::ALL"
874    );
875
876    for &key in Key::ALL {
877        let name = key.name();
878        assert_eq!(
879            Key::from_name(name),
880            Some(key),
881            "Failed to roundtrip {key:?} from name {name:?}"
882        );
883
884        let symbol = key.symbol_or_name();
885        assert_eq!(
886            Key::from_name(symbol),
887            Some(key),
888            "Failed to roundtrip {key:?} from symbol {symbol:?}"
889        );
890    }
