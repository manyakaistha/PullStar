5/// A keyboard shortcut, e.g. `Ctrl+Alt+W`.
6///
7/// Can be used with [`crate::InputState::consume_shortcut`]
8/// and [`crate::Context::format_shortcut`].
9#[derive(Clone, Copy, Debug, Hash, PartialEq, Eq)]
10#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
11pub struct KeyboardShortcut {
12    pub modifiers: Modifiers,
13
14    pub logical_key: Key,
15}
16
17impl KeyboardShortcut {
18    pub const fn new(modifiers: Modifiers, logical_key: Key) -> Self {
19        Self {
20            modifiers,
21            logical_key,
22        }
23    }
24
25    pub fn format(&self, names: &ModifierNames<'_>, is_mac: bool) -> String {
26        let mut s = names.format(&self.modifiers, is_mac);
27        if !s.is_empty() {
28            s += names.concat;
29        }
30        if names.is_short {
31            s += self.logical_key.symbol_or_name();
32        } else {
33            s += self.logical_key.name();
34        }
35        s
36    }
37
38    /// Would consuming this shortcut interfere with editing text
39    /// (e.g. moving the text cursor, or typing a space)?
40    ///
41    /// This is true for any shortcut without modifiers (e.g. `Space`, `A`, `Delete`),
42    /// as well as for shortcuts using keys that are commonly combined with modifiers
43    /// when editing text (e.g. `Alt+ArrowLeft` to move one word, or `Shift+Home` to select to the start of a line).
44    ///
45    /// Such shortcuts should only be consumed when no text field has keyboard focus.
46    /// See also [`crate::Memory::focused`] and [`crate::Context::egui_wants_keyboard_input`].
47    ///
48    /// ```
49    /// # use egui::{Key, KeyboardShortcut, Modifiers};
50    /// assert!(KeyboardShortcut::new(Modifiers::NONE, Key::Space).conflicts_with_text_editing());
51    /// assert!(KeyboardShortcut::new(Modifiers::ALT, Key::ArrowLeft).conflicts_with_text_editing());
52    /// assert!(!KeyboardShortcut::new(Modifiers::COMMAND, Key::S).conflicts_with_text_editing());
53    /// ```
54    pub fn conflicts_with_text_editing(&self) -> bool {
55        self.modifiers.is_none()
56            || matches!(
57                self.logical_key,
58                Key::Space
59                    | Key::ArrowLeft
60                    | Key::ArrowRight
61                    | Key::ArrowUp
62                    | Key::ArrowDown
63                    | Key::Home
64                    | Key::End
65            )
66    }
67}

69#[test]
70fn format_kb_shortcut() {
71    let cmd_shift_f = KeyboardShortcut::new(Modifiers::COMMAND | Modifiers::SHIFT, Key::F);
72    assert_eq!(
73        cmd_shift_f.format(&ModifierNames::NAMES, false),
74        "Ctrl+Shift+F"
75    );
76    assert_eq!(
77        cmd_shift_f.format(&ModifierNames::NAMES, true),
78        "Shift+Cmd+F"
79    );
80    assert_eq!(cmd_shift_f.format(&ModifierNames::SYMBOLS, false), "⌃⇧F");
81    assert_eq!(cmd_shift_f.format(&ModifierNames::SYMBOLS, true), "⇧⌘F");
82}
