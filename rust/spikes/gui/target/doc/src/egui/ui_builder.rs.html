10/// The properties specified when creating a top-level or child [`Ui`].
11///
12/// By default, everything is inherited from the parent,
13/// except for `max_rect` which by default is set to
14/// the parent [`Ui::available_rect_before_wrap`].
15///
16/// See also [`Ui::new`] and [`Ui::new_child`] for uses.
17#[must_use]
18#[derive(Clone, Default)]
19pub struct UiBuilder {
20    pub id_source: Option<IdSource>,
21    pub ui_stack_info: UiStackInfo,
22    pub layer_id: Option<LayerId>,
23    pub max_rect: Option<Rect>,
24    pub layout: Option<Layout>,
25    pub disabled: bool,
26    pub invisible: bool,
27    pub sizing_pass: bool,
28    pub style: Option<Arc<Style>>,
29    pub sense: Option<Sense>,
30    pub accessibility_parent: Option<Id>,
31    pub accessibility_label: Option<String>,
32    pub accessibility_role: Option<crate::accesskit::Role>,
33    pub classes: Classes,
34}

36/// Is this [`Ui`] a root or a child of another [`Ui`]?
37#[derive(Clone)]
38pub enum IdSource {
39    /// Explicitly use this [`Id`]
40    Explicit(Id),
41
42    /// Salt the parent [`Id`] with this.
43    Child(IdSalt),
44}
45
46impl UiBuilder {
47    #[inline]
48    pub fn new() -> Self {
49        Self::default()
50    }
51
52    /// Seed the child `Ui` with this `id_salt`, which will be mixed
53    /// with the [`Ui::scope_id`] of the parent.
54    ///
55    /// You should give each [`Ui`] an `id_salt` that is unique
56    /// within the parent, or give it none at all.
57    #[inline]
58    pub fn id_salt(mut self, id_salt: impl AsIdSalt) -> Self {
59        self.id_source = Some(IdSource::Child(IdSalt::new(id_salt)));
60        self
61    }
62
63    /// Set the [`Ui::scope_id`] of the new `Ui` to something independent of the parent `Ui`.
64    /// This way child widgets can be moved in the ui tree without losing state.
65    /// You have to ensure that in a frame the child widgets do not get rendered in multiple places.
66    ///
67    /// You should set the same unique `scope_id` at every place in the ui tree where you want the
68    /// child widgets to share state.
69    /// If the child widgets are not moved in the ui tree, use [`UiBuilder::id_salt`] instead.
70    ///
71    /// The `scope_id` is also used as the [`Ui::unique_id`] of the new `Ui`, so it must be globally unique.
72    #[inline]
73    pub fn scope_id(mut self, scope_id: Id) -> Self {
74        self.id_source = Some(IdSource::Explicit(scope_id));
75        self
76    }
77
78    /// Renamed to [`Self::scope_id`].
79    #[deprecated = "Renamed to `UiBuilder::scope_id`"]
80    #[inline]
81    pub fn id(self, id: Id) -> Self {
82        self.scope_id(id)
83    }
84
85    /// Provide some information about the new `Ui` being built.
86    #[inline]
87    pub fn ui_stack_info(mut self, ui_stack_info: UiStackInfo) -> Self {
88        self.ui_stack_info = ui_stack_info;
89        self
90    }
91
92    /// Show the [`Ui`] in a different [`LayerId`] from its parent.
93    #[inline]
94    pub fn layer_id(mut self, layer_id: LayerId) -> Self {
95        self.layer_id = Some(layer_id);
96        self
97    }
98
99    /// Set the max rectangle, within which widgets will go.
100    ///
101    /// New widgets will *try* to fit within this rectangle.
102    ///
103    /// Text labels will wrap to fit within `max_rect`.
104    /// Separator lines will span the `max_rect`.
105    ///
106    /// If a new widget doesn't fit within the `max_rect` then the
107    /// [`Ui`] will make room for it by expanding both `min_rect` and
108    ///
109    /// If not set, this will be set to the parent
110    /// [`Ui::available_rect_before_wrap`].
111    #[inline]
112    pub fn max_rect(mut self, max_rect: Rect) -> Self {
113        self.max_rect = Some(max_rect);
114        self
115    }
116
117    /// Override the layout.
118    ///
119    /// Will otherwise be inherited from the parent.
120    #[inline]
121    pub fn layout(mut self, layout: Layout) -> Self {
122        self.layout = Some(layout);
123        self
124    }
125
126    /// Make the new `Ui` disabled, i.e. grayed-out and non-interactive.
127    ///
128    /// Note that if the parent `Ui` is disabled, the child will always be disabled.
129    ///
130    /// See also [`crate::Ui::add_enabled`], [`crate::Ui::add_enabled_ui`] and [`crate::Ui::is_enabled`].
131    #[inline]
132    pub fn disabled(mut self) -> Self {
133        self.disabled = true;
134        self
135    }
136
137    /// Make the contents invisible.
138    ///
139    /// Will also disable the `Ui` (see [`Self::disabled`]).
140    ///
141    /// If the parent `Ui` is invisible, the child will always be invisible.
142    #[inline]
143    pub fn invisible(mut self) -> Self {
144        self.invisible = true;
145        self.disabled = true;
146        self
147    }
148
149    /// Set to true in special cases where we do one frame
150    /// where we size up the contents of the Ui, without actually showing it.
151    ///
152    /// If the `sizing_pass` flag is set on the parent,
153    /// the child will inherit it automatically.
154    #[inline]
155    pub fn sizing_pass(mut self) -> Self {
156        self.sizing_pass = true;
157        self
158    }
159
160    /// Override the style.
161    ///
162    /// Otherwise will inherit the style of the parent.
163    #[inline]
164    pub fn style(mut self, style: impl Into<Arc<Style>>) -> Self {
165        self.style = Some(style.into());
166        self
167    }
168
169    /// Set if you want sense clicks and/or drags. Default is [`Sense::hover`].
170    ///
171    /// The sense will be registered below the Senses of any widgets contained in this [`Ui`], so
172    /// if the user clicks a button contained within this [`Ui`], that button will receive the click
173    /// instead.
174    ///
175    /// The response can be read early with [`Ui::response`].
176    #[inline]
177    pub fn sense(mut self, sense: Sense) -> Self {
178        self.sense = Some(sense);
179        self
180    }
181
182    /// Make this [`Ui`] closable.
183    ///
184    /// Calling [`Ui::close`] in a child [`Ui`] will mark this [`Ui`] for closing.
185    /// After [`Ui::close`] was called, [`Ui::should_close`] and [`crate::Response::should_close`] will
186    /// return `true` (for this frame).
187    ///
188    /// This works by adding a [`ClosableTag`] to the [`UiStackInfo`].
189    #[inline]
190    pub fn closable(mut self) -> Self {
191        self.ui_stack_info
192            .tags
193            .insert(ClosableTag::NAME, Some(Arc::new(ClosableTag::default())));
194        self
195    }
196
197    /// Set the accessibility parent for this [`Ui`].
198    ///
199    /// Pass [`Ui::unique_id`] or `Response::id`, not [`Ui::scope_id`].
200    ///
201    /// This will override the automatic parent assignment for accessibility purposes.
202    /// If not set, the parent [`Ui`]'s ID will be used as the accessibility parent.
203    #[inline]
204    pub fn accessibility_parent(mut self, parent_id: Id) -> Self {
205        self.accessibility_parent = Some(parent_id);
206        self
207    }
208
209    /// Name this [`Ui`] in the accessibility tree.
210    ///
211    /// Every `Ui` is a node there, with a role from its [`UiKind`](crate::UiKind).
212    /// A name lets a screen reader, or a test, tell this panel, popup or region from the others.
213    #[inline]
214    pub fn accessibility_label(mut self, label: impl Into<String>) -> Self {
215        self.accessibility_label = Some(label.into());
216        self
217    }
218
219    /// Give this [`Ui`]'s node a role other than the one its [`UiKind`](crate::UiKind) implies,
220    /// e.g. [`Role::Alert`](crate::accesskit::Role::Alert) for a toast.
221    #[inline]
222    pub fn accessibility_role(mut self, role: crate::accesskit::Role) -> Self {
223        self.accessibility_role = Some(role);
224        self
225    }
226}
