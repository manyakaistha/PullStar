44/// The different kinds of [`crate::Atom`]s.
45#[derive(Default)]
46pub enum AtomKind<'a> {
47    /// Empty, that can be used with [`crate::AtomExt::atom_grow`] to reserve space.
48    #[default]
49    Empty,
50
51    /// Text atom.
52    ///
53    /// Truncation within [`crate::WidgetAtom`] works like this:
54    /// - if `wrap_mode` is not Extend
55    ///   - if no atom is `shrink`
56    ///     - the first text atom is selected and will be marked as `shrink`
57    ///   - the atom marked as `shrink` will shrink / wrap based on the selected wrap mode
58    ///   - any other text atoms will have `wrap_mode` extend
59    /// - if `wrap_mode` is extend, Text will extend as expected.
60    ///
61    /// Unless [`crate::AtomExt::atom_max_width`] is set, `wrap_mode` should only be set via [`crate::Style`] or
62    /// [`crate::ContainerAtom::wrap_mode`], as setting a wrap mode on a [`WidgetText`] atom
63    /// that is not `shrink` will have unexpected results.
64    ///
65    /// The size is determined by converting the [`WidgetText`] into a galley and using the galleys
66    /// size. You can use [`crate::AtomExt::atom_size`] to override this, and [`crate::AtomExt::atom_max_width`]
67    /// to limit the width (Causing the text to wrap or truncate, depending on the `wrap_mode`.
68    /// [`crate::AtomExt::atom_max_height`] has no effect on text.
69    Text(WidgetText),
70
71    /// Image atom.
72    ///
73    /// By default the size is determined via [`Image::calc_size`].
74    /// You can use [`crate::AtomExt::atom_max_size`] or [`crate::AtomExt::atom_size`] to customize the size.
75    /// There is also a helper [`crate::AtomExt::atom_max_height_font_size`] to set the max height to the
76    /// default font height, which is convenient for icons.
77    Image(Image<'a>),
78
79    /// A custom closure that produces a sized atom.
80    ///
81    /// The vec2 passed in is the available size to this atom. The returned vec2 should be the
82    /// preferred / intrinsic size.
83    ///
84    /// Note: This api is experimental, expect breaking changes here.
85    /// When cloning, this will be cloned as [`AtomKind::Empty`].
86    Closure(AtomClosure<'a>),
87
88    /// A closure that paints the atom at the [`Rect`] the layout gives it.
89    ///
90    /// It has no size of its own, so set one with [`crate::AtomExt::atom_size`], or use
91    /// [`crate::Atom::paint`], which does that for you. If the size depends on the [`Ui`],
92    /// return a [`SizedAtomKind::Paint`] from an [`AtomKind::Closure`] instead.
93    ///
94    /// Use this for widgets that draw their own shapes, like the check mark of
95    /// [`crate::Checkbox`].
96    Paint(AtomPaint<'a>),
97
98    /// A nested [`WidgetAtom`], letting you embed an atom-based widget as a single atom
99    /// inside another [`WidgetAtom`].
100    ///
101    /// The nested widget is measured (sized) when the parent is sized, and painted (and
102    /// interacted with) at the cell rect the parent computes for it.
103    ///
104    /// Use [`Self::Container`] instead if you don't need the nested layout to interact.
105    Widget(Box<WidgetAtom<'a>>),
106
107    /// A nested [`ContainerAtom`], letting you embed a non-interactive atom-based layout as a
108    /// single atom inside another [`WidgetAtom`].
109    ///
110    /// Like [`Self::Widget`], the nested layout is measured when the parent is sized and painted
111    /// at the cell rect the parent computes for it. Unlike [`Self::Widget`], a [`ContainerAtom`]
112    /// has no [`Id`](crate::Id) or [`Sense`](crate::Sense), so it is never interacted with.
113    Container(Box<ContainerAtom<'a>>),
114}

116impl Clone for AtomKind<'_> {
117    fn clone(&self) -> Self {
118        match self {
119            AtomKind::Empty => AtomKind::Empty,
120            AtomKind::Text(text) => AtomKind::Text(text.clone()),
121            AtomKind::Image(image) => AtomKind::Image(image.clone()),
122            AtomKind::Closure(_) => {
123                log::warn!("Cannot clone atom closures");
124                AtomKind::Empty
125            }
126            AtomKind::Paint(paint) => AtomKind::Paint(Arc::clone(paint)),
127            AtomKind::Widget(layout) => AtomKind::Widget(layout.clone()),
128            AtomKind::Container(container) => AtomKind::Container(container.clone()),
129        }
130    }
131}

133impl Debug for AtomKind<'_> {
134    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
135        match self {
136            AtomKind::Empty => write!(f, "AtomKind::Empty"),
137            AtomKind::Text(text) => write!(f, "AtomKind::Text({text:?})"),
138            AtomKind::Image(image) => write!(f, "AtomKind::Image({image:?})"),
139            AtomKind::Closure(_) => write!(f, "AtomKind::Closure(<closure>)"),
140            AtomKind::Paint(_) => write!(f, "AtomKind::Paint(<closure>)"),
141            AtomKind::Widget(_) => write!(f, "AtomKind::Widget(<widget>)"),
142            AtomKind::Container(_) => write!(f, "AtomKind::Container(<container>)"),
143        }
144    }
145}

147impl<'a> AtomKind<'a> {
148    /// See [`Self::Text`]
149    pub fn text(text: impl Into<WidgetText>) -> Self {
150        AtomKind::Text(text.into())
151    }
152
153    /// See [`Self::Image`]
154    pub fn image(image: impl Into<Image<'a>>) -> Self {
155        AtomKind::Image(image.into())
156    }
157
158    /// See [`Self::Paint`]
159    pub fn paint(func: impl Fn(&Ui, AtomPaintArgs) + 'a) -> Self {
160        AtomKind::Paint(Arc::new(func))
161    }
162
163    /// See [`Self::Closure`]
164    pub fn closure(func: impl FnOnce(&Ui, IntoSizedArgs) -> IntoSizedResult<'static> + 'a) -> Self {
165        AtomKind::Closure(Box::new(func))
166    }

168    /// Turn this [`AtomKind`] into a [`SizedAtomKind`].
169    ///
170    /// This converts [`WidgetText`] into [`crate::Galley`] and tries to load and size [`Image`].
171    /// The first returned argument is the preferred size.
172    pub fn into_sized(
173        self,
174        ui: &Ui,
175        IntoSizedArgs {
176            available_size,
177            wrap_mode,
178            fallback_font,
179        }: IntoSizedArgs,
180    ) -> IntoSizedResult<'a> {
181        match self {
182            AtomKind::Text(text) => {
183                let galley = text.into_galley(ui, Some(wrap_mode), available_size.x, fallback_font);
184                IntoSizedResult {
185                    intrinsic_size: galley.intrinsic_size(),
186                    sized: SizedAtomKind::Text(galley),
187                }
188            }
189            AtomKind::Image(image) => {
190                let size = image.load_and_calc_size(ui, available_size);
191                let size = size.unwrap_or(Vec2::ZERO);
192                IntoSizedResult {
193                    intrinsic_size: size,
194                    sized: SizedAtomKind::Image { image, size },
195                }
196            }
197            AtomKind::Empty => IntoSizedResult {
198                intrinsic_size: Vec2::ZERO,
199                sized: SizedAtomKind::Empty { size: None },
200            },
201            AtomKind::Closure(func) => func(
202                ui,
203                IntoSizedArgs {
204                    available_size,
205                    wrap_mode,
206                    fallback_font,
207                },
208            ),
209            AtomKind::Paint(paint) => IntoSizedResult {
210                intrinsic_size: Vec2::ZERO,
211                sized: SizedAtomKind::Paint {
212                    paint,
213                    size: Vec2::ZERO,
214                },
215            },
216            AtomKind::Widget(layout) => {
217                let sized = layout.measure(ui, available_size);
218                IntoSizedResult {
219                    intrinsic_size: sized.intrinsic_size,
220                    sized: SizedAtomKind::Widget(Box::new(sized)),
221                }
222            }
223            AtomKind::Container(layout) => {
224                let sized = layout.measure(ui, available_size);
225                IntoSizedResult {
226                    intrinsic_size: sized.intrinsic_size,
227                    sized: SizedAtomKind::Container(Box::new(sized)),
228                }
229            }
230        }
231    }
232}
